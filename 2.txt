// 函数用于将32位网络字节序的IP地址转换为点分十进制字符串  
void int_to_ip(uint32_t ip, char *buffer, size_t buflen) {  
    // 确保buffer足够大以容纳IP地址字符串（包括结尾的空字符）  
    if (buflen < INET_ADDRSTRLEN) {  
        return; // 缓冲区太小，直接返回  
    }  
  
    // 通过位移和掩码提取每个字节，并将其转换为十进制字符串  
    sprintf(buffer, "%d.%d.%d.%d",  
            (ip >> 24) & 0xFF,       // 提取第一个字节  
            (ip >> 16) & 0xFF,       // 提取第二个字节  
            (ip >> 8) & 0xFF,        // 提取第三个字节  
            ip & 0xFF);              // 提取第四个字节  
}

static unsigned int g_serverIpAddr = 0;

float TcpTest_Server_SpeedVerify(int pktNum,int pktLen,int netCardLoc)
{
	int i = 0;
	int ret = 0;
	const int opt = 1;
	int socketFd,connectFd = 0;
	int recvLen = 0,recvPacketLen = 0;
	unsigned int startTime = 0,costTime = 0;
	int serverIpAddr = 0xc0a81267;
	int serverIpAddrBuf[30] = {0};
	float speed = 0.0;
	char * recvBuf = NULL;
	
	recvBuf = (char *)malloc(pktLen);
	if(recvBuf == NULL)
	{
		hklog_err("recvBuf malloc fail\n");
		return -1;
	}
		
	/*TCP服务器框架搭建*/
	socketFd = socket(AF_INET,SOCK_STREAM,0);
	if(socketFd < 0)
	{
		hklog_err("socket create fail\n");
		free(recvBuf);
		return -1;
	}
	
	setsockopt(socketFd, SOL_SOCKET, SO_REUSEPORT, (const char *)&opt, sizeof(opt));
	
	HalConfigInfo_ReadEtherNetAddr(netCardLoc,&serverIpAddr,NULL);
	
	sprintf(serverIpAddrBuf, "%d.%d.%d.%d",  \
			(serverIpAddr >> 24) & 0xFF,     \ 
			(serverIpAddr >> 16) & 0xFF,     \  
			(serverIpAddr >> 8) & 0xFF,      \ 
			serverIpAddr & 0xFF);    
	
	hklog_dbg("server IP addr: %s\n",serverIpAddrBuf);
	
	struct sockaddr_in serverAddr;
	memset(&serverAddr,0,sizeof(serverAddr));
	serverAddr.sin_family = AF_INET;
	serverAddr.sin_addr.s_addr =  inet_addr(serverIpAddrBuf);
	serverAddr.sin_port = htons(NET_PORT_TEST_PORT);
	
	ret = bind(socketFd,(struct sockaddr *)&serverAddr,sizeof(serverAddr));
	if(ret < 0)
	{
		hklog_err("bind fail\n");
		free(recvBuf);
		close(socketFd);
		return -1;
	}
	
	ret = listen(socketFd,8);
	if(ret < 0)
	{
		hklog_err("listen fail\n");
		free(recvBuf);
		close(socketFd);
		return -1;
	}
	
	connectFd = accept(socketFd,NULL,NULL);
	if(connectFd < 0)
	{
		hklog_err("accept fail\n");
		free(recvBuf);
		close(socketFd);
		return -1;
	}
	
	hklog_dbg("have connect a client\n");
	
	while(1)
	{
		 memset(recvBuf,0,sizeof(recvBuf));
		 recvLen = recv(connectFd,recvBuf,pktLen,0);
		 
		 if(recvLen < 0)
		 {
				hklog_err("recv fail\n");
				free(recvBuf);
				close(connectFd);
				close(socketFd);
				return -1;
		 }
		 else if(recvLen == 0)
		 {
				hklog_err("client disconnect\n");
				free(recvBuf);
				close(connectFd);
				close(socketFd);
				return -1;
		 }
		 else
		 { 
			 recvPacketLen += recvLen;
			 
			 if(recvPacketLen == pktLen)
			 {
				 /*收到第一包开始计时*/
				 startTime = SAL_GetTimeUs();
			 }
		 }
		 
		 if(recvPacketLen == pktNum*pktLen)
		 {
			 costTime = SAL_GetTimeDiffUs(startTime);
			 break;
		 }
	}
	
	speed = pktNum*pktLen/(float)costTime;
	hklog_info("Tcp Speed:%0.2fMB/s\n",speed);
	free(recvBuf);
	close(connectFd);
	close(socketFd);
	
	return speed;
}

float TcpTest_Client_SpeedVerify(int pktNum,int pktLen,int netCardLoc)
{
	int i = 0;
	int ret = 0;
	const int opt = 1;
	int socketFd = 0;
	char * sendBuf = NULL;
	int sendLen = 0;
	unsigned int startTime = 0,costTime = 0;
	unsigned int clientIpAddr = 0;
	float speed = 0.0;

	char clientIpAddrBuf[30] = {0};

			(clientIpAddr >> 24) & 0xFF,     \ 
			(clientIpAddr >> 16) & 0xFF,     \  
			(clientIpAddr >> 8) & 0xFF,      \ 
			clientIpAddr & 0xFF);    
	
	hklog_dbg("client IP addr: %s\n",clientIpAddrBuf);
	
	/*获取服务器IP*/
	sprintf(serverIpAddrBuf, "%d.%d.%d.%d",  \
			(g_serverIpAddr >> 24) & 0xFF,     \ 
			(g_serverIpAddr >> 16) & 0xFF,     \  
			(g_serverIpAddr >> 8) & 0xFF,      \ 
			g_serverIpAddr & 0xFF);    
				
	struct sockaddr_in serverAddr,clientAddr;
	memset(&serverAddr,0,sizeof(serverAddr));
	memset(&clientAddr,0,sizeof(clientAddr));
	serverAddr.sin_family = AF_INET;
	serverAddr.sin_addr.s_addr =  inet_addr(serverIpAddrBuf);
	serverAddr.sin_port = htons(NET_PORT_TEST_PORT);
	
	clientAddr.sin_family = AF_INET;
	clientAddr.sin_addr.s_addr =  inet_addr(clientIpAddrBuf);
	clientAddr.sin_port = htons(NET_PORT_TEST_PORT);
	
	ret = bind(socketFd, (struct sockaddr *)&clientAddr, sizeof(clientAddr));
	if(ret < 0)
	{
		hklog_err("bind fail\n");
		close(socketFd);
		free(sendBuf);
		return -1;
	}
	
	ret = connect(socketFd, (struct sockaddr *)&serverAddr, sizeof(serverAddr));
	if(ret < 0)
	{
		hklog_err("connect server fail\n");
		close(socketFd);
		free(sendBuf);
		return -1;
	}
	
	for(i=0;i<pktLen;i++)
	{
		sendBuf[i] = i;
	}
	
	startTime = SAL_GetTimeUs();
	for(i=0;i<pktNum;i++)
	{
		sendLen = send(socketFd,sendBuf,pktLen,0);
		if(sendLen != pktLen)
		{
			hklog_err("send packet %d fail sendLen:%d\n",i,sendBuf[i]);
			close(socketFd);
			free(sendBuf);
			return -1;
		}
	}
	
	costTime = SAL_GetTimeDiffUs(startTime);
	speed = pktNum*pktLen/(float)costTime;
	hklog_info("Tcp Speed:%0.2fMB/s\n",speed);
	free(sendBuf);
	close(socketFd);

	return 0;
}

	int socketFd = 0;

	float udp_loss_rate = 0.0;
	int serverAddrSize = 0;

	socketFd = socket(AF_INET,SOCK_DGRAM,0);

			(g_serverIpAddr >> 16) & 0xFF,     \  
			(g_serverIpAddr >> 8) & 0xFF,      \

	memset(&serverAddr,0,sizeof(serverAddr));
	memset(&clientAddr,0,sizeof(clientAddr));
	
	serverAddr.sin_family = AF_INET;
	serverAddr.sin_addr.s_addr =  inet_addr(serverIpAddrBuf);
	serverAddr.sin_port = htons(NET_PORT_TEST_PORT);
	
	clientAddr.sin_family = AF_INET;
	clientAddr.sin_addr.s_addr =  inet_addr(clientIpAddrBuf);

		FD_ZERO(&writeFds);
	
		/*把套接字加入读写fd_set集合，FD_SET*/

		tv.tv_sec = 5;
		tv.tv_usec = 0;
		
		/*select检查套接字是否可写*/

		{
			if(FD_ISSET(socketFd,&writeFds))

				sendto(socketFd, sendBuf, pktLen, 0, (SOCKADDR*)&serverAddr, sizeof(serverAddr));
				
				FD_ZERO(&readFds);
				
				FD_SET(socketFd,&readFds);
				
				/*设置超时时间*/
				tv.tv_sec = 5;
				tv.tv_usec = 0;
				
				ret = select(socketFd+1,&readFds,NULL,NULL,&tv);					
				if(ret < 0)
				{
					hklog_info("socket recv  %d pakcet select error\n",i);
					close(socketFd);
					free(sendBuf);
					free(recvBuf);
					return -1;				
				}
				else if(ret == 0)
				{
					hklog_info("socket recv %d pakcet time out \n",i);
				}
				else
				{
					if(FD_ISSET(socketFd,&readFds))
					{
						recvLen = recvfrom(socketFd, recvBuf, pktLen, 0, NULL, NULL);
						if(recvBuf[0] == sendBuf[0])
						{
							udp_recv_num++;
							//hklog_info("correct pktNum %d\n",i);
						}
						else
						{
							hklog_info("error pktNum %d:recvBuf[0]=%d sendBuf[0]=%d\n",i,recvBuf[0],sendBuf[0]);
						}
						
					}
				}
			}
		}
	}
	
	sendBuf[4] = 0xaa;
	
	/*使用FD_ZERO 清空fd_set*/
	FD_ZERO(&writeFds);

	/*把套接字加入读写fd_set集合，FD_SET*/
	FD_SET(socketFd,&writeFds);
	
	/*设置超时时间*/
	tv.tv_sec = 5;
	tv.tv_usec = 0;
	
	/*select检查套接字是否可写*/
	ret = select(socketFd+1,NULL,&writeFds,NULL,&tv);
	
	if(ret > 0)
	{
		if(FD_ISSET(socketFd,&writeFds))
		{
			sendto(socketFd, sendBuf, pktLen, 0, (SOCKADDR*)&serverAddr, sizeof(serverAddr));
		}
	}
	

	udp_loss = pktNum - udp_recv_num;
	udp_loss_rate = udp_loss/pktNum;
	hklog_info("udp_recv_num=%d,udp_loss=%d,pktNum=%d,udp_loss_rate = %0.2f%%\n",udp_recv_num,udp_loss,pktNum,udp_loss_rate*100);
	close(socketFd);
	free(sendBuf);
	free(recvBuf);
	return 0;
}
